/* =========================================================================================
 * 2027 Team_Dynamics Dual APPS & Soft BSPD Plausibility Check (KSAE Rule Compliant)
 * Target MCU: STM32F767ZI | Core Clock: 216MHz | Execution Cycle: 2ms (500Hz, TIM3)
 * ========================================================================================= */
typedef enum {
    APPS_STATE_IDLE = 0,
    APPS_STATE_NORMAL_DRIVE,
    APPS_STATE_IMPLAUSIBLE,
    APPS_STATE_FAULT_LOCKOUT
} APPS_FSM_State_t;

static APPS_FSM_State_t g_apps_state = APPS_STATE_IDLE;
static uint32_t g_implausible_timer_ms = 0;

void APPS_Plausibility_Task_2ms(float raw_apps1_v, float raw_apps2_v, float brake_bar, float *out_torque_cmd) {
    /* 1. Range Check (0.5V~4.5V normal, <0.35V or >4.65V Wire Short/Break Fault) */
    if (raw_apps1_v < 0.35f || raw_apps1_v > 4.65f || raw_apps2_v < 0.35f || raw_apps2_v > 4.65f) {
        g_apps_state = APPS_STATE_FAULT_LOCKOUT;
        *out_torque_cmd = 0.0f;
        return;
    }

    /* 2. Stroke Normalization (APPS1: Positive slope 0.5->4.5V, APPS2: Negative slope 4.5->0.5V) */
    float stroke1 = (raw_apps1_v - 0.50f) / (4.50f - 0.50f) * 100.0f;
    float stroke2 = (4.50f - raw_apps2_v) / (4.50f - 0.50f) * 100.0f;
    if (stroke1 < 0.0f) stroke1 = 0.0f; else if (stroke1 > 100.0f) stroke1 = 100.0f;
    if (stroke2 < 0.0f) stroke2 = 0.0f; else if (stroke2 > 100.0f) stroke2 = 100.0f;
    float stroke_diff = fabsf(stroke1 - stroke2);

    /* 3. Finite State Machine (10% Implausibility for >100ms Rule) */
    switch (g_apps_state) {
        case APPS_STATE_IDLE:
        case APPS_STATE_NORMAL_DRIVE:
            if (stroke_diff > 10.0f) {
                g_apps_state = APPS_STATE_IMPLAUSIBLE;
                g_implausible_timer_ms = 0;
            } else {
                g_apps_state = (stroke1 > 3.0f) ? APPS_STATE_NORMAL_DRIVE : APPS_STATE_IDLE;
            }
            break;

        case APPS_STATE_IMPLAUSIBLE:
            if (stroke_diff <= 10.0f) {
                g_apps_state = APPS_STATE_NORMAL_DRIVE;
                g_implausible_timer_ms = 0;
            } else {
                g_implausible_timer_ms += 2;
                if (g_implausible_timer_ms >= 100) { /* >100ms Latch Lockout */
                    g_apps_state = APPS_STATE_FAULT_LOCKOUT;
                }
            }
            break;

        case APPS_STATE_FAULT_LOCKOUT:
            /* Latch persists until pedal fully released (<5%) and deviation cleared */
            if (stroke1 < 5.0f && stroke2 < 5.0f && stroke_diff < 5.0f) {
                g_apps_state = APPS_STATE_IDLE;
            }
            *out_torque_cmd = 0.0f;
            return;
    }

    /* 4. Software BSPD Pre-Cutoff (Brake > 15 bar && Throttle > 25% for 20ms -> Soft Cut to 0 Nm) */
    static uint32_t soft_bspd_timer_ms = 0;
    if (brake_bar > 15.0f && stroke1 > 25.0f) {
        soft_bspd_timer_ms += 2;
        if (soft_bspd_timer_ms >= 20) {
            *out_torque_cmd = 0.0f;
            return;
        }
    } else {
        soft_bspd_timer_ms = 0;
    }

    /* 5. Safe Base Torque Command Output (Mapped to 0~50 N.m per motor) */
    *out_torque_cmd = (stroke1 / 100.0f) * 50.0f;
}

/* =========================================================================================
 * CAN-FD Dual Inverter Torque Command Transmission (200Hz, 5ms Interval)
 * CAN ID: 0x101 (Left ND72530), 0x102 (Right ND72530) | Baud: 1Mbps
 * ========================================================================================= */
void CAN_Send_Inverter_Command(CAN_HandleTypeDef *hcan, float t_left_nm, float t_right_nm, uint16_t max_rpm) {
    CAN_TxHeaderTypeDef tx_header;
    uint8_t tx_data[8];
    uint32_t tx_mailbox;

    tx_header.IDE = CAN_ID_STD;
    tx_header.RTR = CAN_RTR_DATA;
    tx_header.DLC = 8;

    /* Left Motor Packet (0x101) */
    uint16_t raw_t_left = (uint16_t)(t_left_nm * 10.0f); /* 0.1 N.m / LSB */
    tx_header.StdId = 0x101;
    tx_data[0] = 0x01;                           /* Control Word: Enable Drive */
    tx_data[1] = (uint8_t)(raw_t_left >> 8);     /* Target Torque MSB */
    tx_data[2] = (uint8_t)(raw_t_left & 0xFF);   /* Target Torque LSB */
    tx_data[3] = (uint8_t)(max_rpm >> 8);        /* Max Speed Limit MSB */
    tx_data[4] = (uint8_t)(max_rpm & 0xFF);      /* Max Speed Limit LSB */
    tx_data[5] = 0x52;                           /* Line Current Limit: 82A */
    tx_data[6] = 0x00; tx_data[7] = 0x00;
    HAL_CAN_AddTxMessage(hcan, &tx_header, tx_data, &tx_mailbox);

    /* Right Motor Packet (0x102) */
    uint16_t raw_t_right = (uint16_t)(t_right_nm * 10.0f);
    tx_header.StdId = 0x102;
    tx_data[1] = (uint8_t)(raw_t_right >> 8);
    tx_data[2] = (uint8_t)(raw_t_right & 0xFF);
    HAL_CAN_AddTxMessage(hcan, &tx_header, tx_data, &tx_mailbox);
}
